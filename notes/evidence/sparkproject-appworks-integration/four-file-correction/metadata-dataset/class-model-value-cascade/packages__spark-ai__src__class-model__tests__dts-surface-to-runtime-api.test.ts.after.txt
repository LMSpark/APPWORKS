import { mkdirSync, readFileSync, rmSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { dirname, resolve } from 'node:path'
import { fileURLToPath, pathToFileURL } from 'node:url'
import { describe, expect, it } from 'vitest'

import { buildDtsClassModelBundle } from '../class-model/bundle/build-dts-class-model-bundle'
import { DtsClassModelBundleLoader } from '../class-model/bundle/dts-class-model-bundle-loader'
import { createRuntimeApiMetadataFromSurface } from '../class-model/dts-surface-to-runtime-api'
import { readDtsFileProjectionDocument } from '../class-model/bundle/read-dts-class-model-bundle-json'
import { executeDtsNativeScript } from '../../agent/native-runtime/dts-native-script-runner'
import { ClassModelRuntimeSchema } from '../class-model/schema/class-model-runtime-schema'
import { JsonSchemaValidator } from '@spark-appworks/spark-json-document'

describe('dts-surface-to-runtime-api', () => {
  it('preserves local definition scopes and literal data while collecting schema references', () => {
    const binder = new ClassModelRuntimeSchema({schemaVersion: 1, source: 'class-model-emit', configPath: '', models: {}, fileIndex: {}})
    const schema = binder.bind({type: 'object', properties: {child: {$ref: '#/$defs/Child'}}, $defs: {
      Shared: {type: 'string'},
      Parent: {type: 'object', properties: {value: {$ref: '#/$defs/Shared'}}},
      Child: {type: 'object', $defs: {Shared: {type: 'number'}}, properties: {
        own: {$ref: '#/$defs/Shared'}, inherited: {$ref: '#/$defs/Parent'},
      }},
    }, default: {$ref: 'literal data'}})
    expect(JsonSchemaValidator.validateDeserializedParams({child: {own: 1, inherited: {value: 'a'}}}, schema).ok).toBe(true)
    expect(JsonSchemaValidator.validateDeserializedParams({child: {own: '1', inherited: {value: 'a'}}}, schema).ok).toBe(false)
    expect(schema.default).toEqual({$ref: 'literal data'})
    expect(() => binder.bind({type: 'object', properties: {missing: {$ref: 'unknown.json#/$defs/Missing'}}})).toThrow('Missing ClassModel schema definition')
  })

  it('executes and validates an object argument across shards, including recursive values', async () => {
    const tempRoot = resolve(tmpdir(), `spark-dts-object-runtime-${String(process.pid)}-${String(Date.now())}`)
    const sourceDir = resolve(tempRoot, 'class-model-emit/packages/spark-ai/src/tests')
    const inputPath = resolve(sourceDir, 'selection-input.d.ts')
    const rootPath = resolve(sourceDir, 'selection-owner.d.ts')
    try {
      mkdirSync(sourceDir, {recursive: true})
      writeFileSync(inputPath, [
        'export interface SelectionInput {',
        "  valueFormat: 'native' | 'selection-string'",
        '  value: string',
        '  next?: SelectionInput',
        '}',
      ].join('\n'))
      writeFileSync(rootPath, [
        "import type {SelectionInput} from './selection-input'",
        'export class SelectionOwner { save(params: SelectionInput): string }',
      ].join('\n'))
      const bundle = buildDtsClassModelBundle({repoRoot: tempRoot, rootFiles: [rootPath, inputPath],
        outputDir: resolve(tempRoot, 'generated/dts-class-model')})
      const calls: unknown[] = []
      const instance = {save: (params: unknown) => {calls.push(params); return 'saved'}}
      const command = {instance, rootClassName: 'SelectionOwner', manifestUrl: pathToFileURL(bundle.manifestPath).href,
        fetchJson: async (url: string): Promise<unknown> => JSON.parse(readFileSync(fileURLToPath(url), 'utf8'))}
      const input = {valueFormat: 'selection-string', value: '01|0', next: {valueFormat: 'native', value: 'A,B'}}
      const valid = await executeDtsNativeScript({...command, script: `return await this.save(${JSON.stringify(input)})`})
      expect(valid.checks ?? []).toEqual([])
      expect(valid).toMatchObject({ok: true, data: 'saved'})
      expect(calls).toEqual([input])
      for (const invalid of [
        {valueFormat: 'guess', value: '01|0'},
        {valueFormat: 'selection-string'},
        {...input, next: {valueFormat: 'native', value: 9}},
      ]) {
        const rejected = await executeDtsNativeScript({...command, script: `return await this.save(${JSON.stringify(invalid)})`})
        expect(rejected).toMatchObject({ok: false, checks: expect.arrayContaining([
          expect.objectContaining({code: 'SCHEMA_VALIDATION_FAILED'}),
        ])})
      }
      expect(calls).toHaveLength(1)
    } finally {
      if (dirname(tempRoot) !== resolve(tmpdir())) throw new Error('unsafe test cleanup path')
      rmSync(tempRoot, {recursive: true, force: true})
    }
  })

  it('maps guide bundle shard paramsSchema into script runtime API metadata', async () => {
    const tempRoot = resolve(tmpdir(), `spark-dts-surface-runtime-api-${String(process.pid)}-${String(Date.now())}`)
    const sourcePath = 'class-model-emit/packages/spark-ai/src/tests/script-contract.d.ts'
    const absolutePath = resolve(tempRoot, sourcePath)
    const outputDir = resolve(tempRoot, 'generated/dts-class-model')
    try {
      mkdirSync(dirname(absolutePath), { recursive: true })
      writeFileSync(absolutePath, [
        '/** Script contract root. */',
        'export class ScriptContractRoot {',
        '  /** Opens a child by id. */',
        '  openChild(childId: string): ScriptContractChild',
        '}',
        '/** Script contract child. */',
        'export class ScriptContractChild {',
        '  /** Reads the child label. */',
        '  readLabel(prefix: string): string',
        '}',
      ].join('\n'), 'utf8')

      const bundle = buildDtsClassModelBundle({
        repoRoot: tempRoot,
        rootFiles: [absolutePath],
        outputDir,
      })
      const manifestEntry = bundle.manifest.classIndex['ScriptContractRoot']
      expect(manifestEntry).toBeDefined()
      const shard = readDtsFileProjectionDocument(
        JSON.parse(readFileSync(resolve(outputDir, manifestEntry!.file), 'utf8')) as unknown,
      )
      const scriptContractRoot = shard.models['ScriptContractRoot']
      if (scriptContractRoot?.declarationKind !== 'class') throw new Error('expected class model')
      const openChild = scriptContractRoot.classDecl.members.methods.find(method => method.name === 'openChild')
      expect(openChild?.paramsSchema).toMatchObject({
        type: 'object',
        properties: {
          childId: { type: 'string' },
        },
      })

      const loader = new DtsClassModelBundleLoader({
        manifestUrl: pathToFileURL(bundle.manifestPath).href,
        fetchJson: async url => JSON.parse(readFileSync(fileURLToPath(url), 'utf8')) as unknown,
      })
      const fromLoader = await loader.buildRuntimeApiMetadata('ScriptContractRoot')
      const fromSurface = createRuntimeApiMetadataFromSurface(loader.buildLoadedSurface(), 'ScriptContractRoot')

      const openChildAction = fromLoader.rootApi.actions.find(action => action.name === 'openChild')
      const surfaceOpenChildAction = fromSurface.rootApi.actions.find(action => action.name === 'openChild')
      expect(openChildAction?.paramsSchema).toEqual(openChild?.paramsSchema)
      expect(surfaceOpenChildAction?.paramsSchema).toEqual(openChild?.paramsSchema)
      expect(fromLoader.apiRegistry?.['ScriptContractChild']).toBeDefined()
      expect(loader.buildLoadedSurface().models['ScriptContractChild']).toBeDefined()
    } finally {
      rmSync(tempRoot, { recursive: true, force: true })
    }
  })
})
