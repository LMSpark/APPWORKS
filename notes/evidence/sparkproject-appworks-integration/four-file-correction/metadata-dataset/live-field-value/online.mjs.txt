import fs from 'node:fs'
import assert from 'node:assert/strict'
import {createHash} from 'node:crypto'
import {createServer} from 'vite'

for (const level of ['log', 'info', 'debug', 'warn', 'error']) console[level] = () => {}
const appId = 'D99A1DCE9894698799101EFD70F8FC76'
const targetId = '97DCB03F75AADEAE6B102B062DB71CEA'
const metadataId = '8D1AB14DD8277F3E7017CD38F77B09FD'
const report = {at: new Date().toISOString(), stage: 'initialize', writes: 0, reads: 0, files: [], selections: []}
const hash = text => createHash('sha256').update(text).digest('hex')
let server, session, candidate
try {
  const auth = JSON.parse(fs.readFileSync(0, 'utf8').replace(/^\uFEFF/, ''))
  const storage = new Map()
  Object.defineProperty(globalThis, 'localStorage', {configurable: true, value: {
    getItem: key => storage.get(key) ?? null, setItem: (key, value) => storage.set(key, value),
    removeItem: key => storage.delete(key),
  }})
  server = await createServer({configFile: 'vitest.config.ts', logLevel: 'silent', server: {middlewareMode: true}})
  const {lowcodeApi: api, lowcodeHttp: http, createLowcodeProjectGateways} = await server.ssrLoadModule('/src/lowcode/lowcode-runtime.ts')
  const {openLowcodeDataSpaceDesignSession} = await server.ssrLoadModule('/src/lowcode/data-space/lowcode-data-space-design.ts')
  const {ScenarioViewConfig} = await server.ssrLoadModule('/packages/spark-project-model/src/scenario/scenario-view-config.ts')
  const {DataMember, resolveDataViewMember} = await server.ssrLoadModule('/packages/spark-data/src/core/data-view-key.ts')
  const {RequestState} = await server.ssrLoadModule('/packages/spark-data/src/types.ts')
  let authenticated = false
  http.interceptors.request.use({onRequest: config => {
    if (authenticated) {
      assert.ok(['/api/DataOperation/GetData', '/api/File/content/text'].includes(config.url), 'READ_ONLY_PROBE: unexpected endpoint')
      report.reads++
    }
    return {...config, baseURL: 'http://127.0.0.1:5273'}
  }})
  report.stage = 'login'
  const enterprises = (await api.platform.listEnterprises()).filter(item => item.name === auth.enterpriseName || item.shortName === auth.enterpriseName)
  assert.equal(enterprises.length, 1)
  auth.enterpriseName = enterprises[0].shortName
  await api.platform.login(auth)
  auth.password = ''
  await api.platform.activateApplication(appId)
  authenticated = true
  const gateway = createLowcodeProjectGateways(appId).scenarioViews
  report.stage = 'inspect-persisted-cascades'
  const remote = new Map()
  for (const id of [targetId, metadataId]) {
    const text = await gateway.readText(id)
    remote.set(id, text)
    const parsed = text === null ? null : JSON.parse(text)
    let validation = 'missing'
    if (text !== null) {
      try {new ScenarioViewConfig(id, text); validation = 'valid'} catch (error) {validation = String(error.message).slice(0, 200)}
    }
    report.files.push({scenarioId: id, present: text !== null, hash: text === null ? null : hash(text), validation,
      cascades: (parsed?.viewCascades ?? []).map(item => ({kind: item.kind ?? 'query', keys: Object.keys(item), valueFormat: item.valueFormat ?? null})),
      modelCount: Object.keys(parsed?.tables ?? {}).length})
  }
  report.stage = 'single-target-dataset'
  session = await openLowcodeDataSpaceDesignSession(targetId)
  assert.equal(session.definitionError, null)
  assert.equal(session.metadataDataSet.scenarioId, metadataId)
  assert.equal(session.definitionDataSet.scenarioId, targetId)
  const metadata = session.metadataDataSet
  report.metadata = Object.entries(metadata.tables).map(([name, table]) => {
    const view = table.views.default
    assert.equal(view.rows.length, view.total)
    assert.ok(view.rows.every(row => row[name === 'Base_DataSet' ? 'rowid' : 'dataSetId'] === targetId))
    return {name, rows: view.rows.length, views: Object.keys(table.views)}
  })
  report.definition = Object.entries(session.definitionDataSet.tables).map(([name, table]) => ({
    name, columns: table.columns.filter(column => !column.isComputed).length, views: Object.keys(table.views),
  }))
  report.stage = 'field-permission-and-native-value'
  candidate = metadata
  const parent = candidate.getView('Base_DataModel','modelOptions')
  const target = candidate.getView('Base_DataSet','default')
  const options = candidate.getView('Base_DataModel_Field','apiInfoFields')
  const allFields = candidate.getView('Base_DataModel_Field','default').rows
  await parent.loadFromServer({allPages:true})
  const models = parent.rows.filter(row=>allFields.some(field=>field.dataModelId===parent.getPkKey(row)))
  assert.ok(models.length>=2)
  const [first,second]=models
  const targetRow=target.rows.find(row=>target.getPkKey(row)===targetId)
  assert.ok(targetRow)
  target.setCurrentRowById(targetId)
  report.targetPermission={table:'Base_DataSet',field:'Name',access:target.fieldAccess(targetRow,'Name')}
  assert.equal(report.targetPermission.access.read,'visible')
  assert.equal(report.targetPermission.access.write,'allowed')
  const original=targetRow.Name
  assert.equal(typeof original,'string')
  const token1=allFields.find(row=>row.dataModelId===parent.getPkKey(first)).rowid
  const token2=allFields.find(row=>row.dataModelId===parent.getPkKey(second)).rowid
  assert.ok(typeof token1==='string'&&typeof token2==='string'&&token1!==token2)
  options.valueField='rowid'
  options.labelField='Name'
  options.selectionDelimiter='|'
  options.filterExpression={logic:'and',filters:[options.filterExpression,
    {field:'dataModelId',operator:'in',value:{Type:'GetInputParam',ParamName:'modelIds'}},
    {field:'dataSetId',operator:'eq',value:{Type:'GetInputParam',ParamName:'spaceId'}}]}
  parent.setSelectedRows([])
  const definition={kind:'field',cascadeId:'online-field-value',tableName:'Base_DataSet',viewId:'default',targetField:'Name',
    valueFormat:'selection-string',parents:[{tableName:'Base_DataModel',viewId:'modelOptions',parameter:'modelIds'},
      {tableName:'Base_DataSet',viewId:'default',field:'rowid',parameter:'spaceId'}],
    optionsView:{tableName:'Base_DataModel_Field',viewId:'apiInfoFields'},valuePolicy:{mode:'retain-valid',clearValue:''}}
  candidate.addCascade(definition)
  const address={tableName:'Base_DataSet',viewId:'default',field:'Name'}
  target.updateEditingValue(targetId,'Name',[token1,token2,'invalid-verification-token'].join('|'))
  report.stage='actual-option-query-and-local-target-update'
  for(const [selected,expected] of [[[first,second],[token1,token2].join('|')],[[second],token2],[[], '']]) {
    parent.setSelectedRows(selected)
    const expectedModels=selected.map(row=>parent.getPkKey(row))
    const expectedOptions=allFields.filter(row=>expectedModels.includes(row.dataModelId)).map(row=>row.rowid).sort()
    const deadline=Date.now()+10000
    let state
    while(true) {
      state=candidate.getFieldCascadeState(address)
      if(state.status==='error') throw new Error(state.error)
      if(state.status==='ready'&&target.getEditingRow(targetId).Name===expected) break
      if(Date.now()>deadline) throw new Error(`Field cascade did not converge ${state.status}/${state.valueStatus}`)
      await new Promise(resolve=>setTimeout(resolve,25))
    }
    assert.deepEqual(state.options.map(row=>row.rowid).sort(),expectedOptions)
    assert.equal(targetRow.Name,original)
    assert.equal(typeof target.getEditingRow(targetId).Name,'string')
    assert.equal(options.rows.length,0)
    report.selections.push({models:selected.length,options:state.options.length,retainedTokens:expected===''?0:expected.split('|').length,
      valueStatus:state.valueStatus,originalRowsUnchanged:true,overlayString:true})
  }
  report.stage='independent-second-parent-value'
  candidate.removeCascade({cascadeId:'online-field-value'})
  const pointerParent=candidate.getView('Base_DataModel','default')
  pointerParent.setCurrentRowById(parent.getPkKey(first))
  parent.setSelectedRows([first,second])
  options.filterExpression={logic:'and',filters:[
    {field:'dataSetId',operator:'eq',value:{Type:'GetInputParam',ParamName:'formid'}},
    {field:'dataModelId',operator:'in',value:{Type:'GetInputParam',ParamName:'modelIds'}},
    {field:'dataModelId',operator:'eq',value:{Type:'GetInputParam',ParamName:'activeModelId'}}]}
  target.updateEditingValue(targetId,'Name',[token1,token2,'invalid-verification-token'].join('|'))
  candidate.addCascade({...definition,parents:[definition.parents[0],
    {tableName:'Base_DataModel',viewId:'default',field:'rowid',parameter:'activeModelId'}]})
  report.secondParent=[]
  for(const [pointer,expected] of [[second,token2],[first,'']]) {
    pointerParent.setCurrentRowById(parent.getPkKey(pointer))
    const expectedOptions=allFields.filter(row=>row.dataModelId===parent.getPkKey(pointer)).map(row=>row.rowid).sort()
    const deadline=Date.now()+10000
    let state
    while(true) {
      state=candidate.getFieldCascadeState(address)
      if(state.status==='error') throw new Error(state.error)
      if(state.status==='ready'&&target.getEditingRow(targetId).Name===expected) break
      if(Date.now()>deadline) throw new Error(`Second input not converged ${state.status}/${state.valueStatus}`)
      await new Promise(resolve=>setTimeout(resolve,25))
    }
    assert.deepEqual(state.options.map(row=>row.rowid).sort(),expectedOptions)
    assert.equal(parent.selectedRows.length,2)
    assert.equal(targetRow.Name,original)
    report.secondParent.push({unchangedSelectionCount:2,options:state.options.length,retainedTokens:expected===''?0:1,valueStatus:state.valueStatus})
  }
  const readsBefore=report.reads
  pointerParent.setCurrentRowById(parent.getPkKey(first))
  await new Promise(resolve=>setTimeout(resolve,100))
  assert.equal(report.reads,readsBefore)
  report.sameScalarValueNoRequest=true
  target.discardEditingRows([targetId])
  assert.equal(target.getEditingRow(targetId).Name,original)
  assert.equal(target.getEditingPatch(targetId),undefined)
  report.cancelRestored=true
  candidate.removeCascade({cascadeId:'online-field-value'})
  const verifySession=await openLowcodeDataSpaceDesignSession(targetId)
  try {
    assert.equal(verifySession.metadataDataSet.getView('Base_DataSet','default').rows[0].Name,original)
    report.independentDatabaseReadUnchanged=true
  } finally {verifySession.dispose()}
  for (const [id, before] of remote) assert.equal(await gateway.readText(id), before)

  report.remoteUnchanged = true
  report.candidateOnly = true
  report.stage = 'complete'
} catch (error) {
  report.failed = true
  report.error = String(error.message ?? error).replace(/Bearer\s+\S+/gi, '[redacted]').replace(/eyJ[A-Za-z0-9_.-]+/g, '[redacted]').slice(0, 260)
  process.exitCode = 1
} finally {
  candidate?.destroy()
  session?.dispose()
  await server?.close()
  fs.writeFileSync(new URL('./online-result.json', import.meta.url), JSON.stringify(report, null, 2))
  process.stdout.write(JSON.stringify(report, null, 2))
}
