function deferred<T>(): Readonly<{
  promise: Promise<T>
  resolve: (value: T) => void
  reject: (reason: unknown) => void
}> {
  let resolve: (value: T) => void = () => undefined
  let reject: (reason: unknown) => void = () => undefined
  const promise = new Promise<T>((resolvePromise, rejectPromise) => {
    resolve = resolvePromise
    reject = rejectPromise
  })
  return { promise, resolve, reject }
}

  return wrapper

const LifecyclePage = defineComponent({
  setup() {
    const lifecycle = useSystemPageIdentityLifecycle()
    revokedLifecycle = lifecycle
    const dirty = ref(false)
    const submitting = ref(false)
    const aborted = ref(lifecycle.signal.aborted)
    lifecycle.signal.addEventListener('abort', () => { aborted.value = true })
    return () => h('div', [h('button', {
      'data-aborted': aborted.value,
      disabled: !lifecycle.editable,
      onClick: () => {
        const next = !dirty.value
        try { lifecycle.reportDirty(next); dirty.value = next; lifecycleReportError = null }
        catch (error: unknown) { lifecycleReportError = error instanceof Error ? error.message : String(error) }
      },
    }, dirty.value ? 'dirty' : 'clean'), h('button', { onClick: () => {
      const next = !submitting.value
      try { lifecycle.reportSubmitting(next); submitting.value = next; lifecycleReportError = null }
      catch (error: unknown) { lifecycleReportError = error instanceof Error ? error.message : String(error) }
    } }, submitting.value ? 'submitting' : 'idle')])
  },
})

async function setupIdentity(component = page, includeThirdNode = false) {
  const router = createRouter({ history: createMemoryHistory(), routes: [] })
  let failNavigationLoad = false
  let navigation = {
    id: 'root', title: 'Root', childPlacement: 'header' as const, projectId: 'project',
    items: [
        { id: 'system-node', title: 'Dashboard', path: '/dashboard' },
      { id: 'system-node-two', title: 'Dashboard Two', itemKind: 'system-page' as const, path: '/dashboard' },
      ...(includeThirdNode ? [{ id: 'system-node-three', title: 'Dashboard Three', itemKind: 'system-page' as const, path: '/dashboard' }] : []),
    ],
  }
  const dynamic = createDynamicRouter({
    router,
    pageComponent: page,
    tenantPathPrefix: '/t/:tenantId/:projectId',
    loadNavigation: async () => { if (failNavigationLoad) throw new Error('navigation load failed'); return navigation },
    componentMap: { '/dashboard': component },
  })
  await dynamic.registerRoutes()
  return { router, dynamic, setNavigation: (next: typeof navigation) => { navigation = next },
    setFailNavigationLoad: (next: boolean) => { failNavigationLoad = next } }
}
